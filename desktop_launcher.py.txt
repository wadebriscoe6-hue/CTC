"""
Cross-Platform Desktop Launcher for Cloth Talk AI
Launches the local Streamlit server on your PC/Mac and automatically opens your default browser.
"""

import os
import sys
import subprocess
import time
import webbrowser

def main():
    app_dir = os.path.dirname(os.path.abspath(__file__))
    app_py = os.path.join(app_dir, "app.py")
    
    print("=" * 70)
    print("  💎 CLOTH TALK AI - DESKTOP LOCAL HOST LAUNCHER")
    print("  Branded for @clothtalk88 | Jarvis Diamond Protocol")
    print("=" * 70)
    print("\n[*] Starting local application server on port 8501...")
    
    port = 8501
    url = f"http://localhost:{port}"

    def open_browser():
        time.sleep(1.8)
        print(f"[*] Opening browser: {url}")
        webbrowser.open(url)

    import threading
    threading.Thread(target=open_browser, daemon=True).start()

    cmd = [
        sys.executable,
        "-m",
        "streamlit",
        "run",
        app_py,
        f"--server.port={port}",
        "--server.headless=false",
        "--theme.base=dark"
    ]
    
    try:
        subprocess.run(cmd)
    except KeyboardInterrupt:
        print("\n[*] Desktop server stopped. Have a great day!")

if __name__ == "__main__":
    main()
