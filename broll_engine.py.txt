    def get_recommended_broll(self, transcript_text: str) -> List[Dict[str, Any]]:
        txt = transcript_text.lower()
        scored_broll = []
        for item in self.BROLL_LIBRARY:
            score = 10
            if "diamond" in txt and "diamond" in item["id"]: score += 30
            if any(w in txt for w in ["legacy", "family", "father", "son"]) and "legacy" in item["id"]: score += 35
            if any(w in txt for w in ["discipline", "strength", "punch", "fight"]) and "boxing" in item["id"]: score += 25
            if any(w in txt for w in ["cloth talk", "listen", "studio"]) and "studio" in item["id"]: score += 25
            if any(w in txt for w in ["matrix", "code", "system"]) and "matrix" in item["id"]: score += 25

            item_copy = dict(item)
            item_copy["relevance_score"] = score
            scored_broll.append(item_copy)

        scored_broll.sort(key=lambda x: x["relevance_score"], reverse=True)
        return scored_broll
