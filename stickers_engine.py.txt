    def auto_place_stickers(
        self, 
        transcript_sentences: List[Dict[str, Any]], 
        duration_seconds: float, 
        density: str = "Medium"
    ) -> List[Dict[str, Any]]:
        if density == "Off": return []

        placed_stickers = []
        full_text = " ".join([s.get("text", "") for s in transcript_sentences]).lower()

        # 1. Hook Opener Sticker
        hook_candidates = [s for s in self.FREE_STICKERS_CATALOG if any(kw in full_text[:60] for kw in s["trigger_keywords"])]
        hook_stk = hook_candidates[0] if hook_candidates else self.FREE_STICKERS_CATALOG[5]
        placed_stickers.append({
            "sticker": hook_stk,
            "start_time": 0.5,
            "end_time": min(4.0, duration_seconds * 0.3),
            "position": "top-right"
        })

        # 2. Outro Follow CTA
        outro_stk = self.FREE_STICKERS_CATALOG[0]
        placed_stickers.append({
            "sticker": outro_stk,
            "start_time": max(2.0, duration_seconds - 6.0),
            "end_time": duration_seconds,
            "position": "center-float"
        })

        # 3. Mid-Roll Stickers
        if density in ["Medium", "High"]:
            mid_interval = 7.0 if density == "High" else 14.0
            curr_t = 6.0
            while curr_t < (duration_seconds - 7.0):
                avail = [s for s in self.FREE_STICKERS_CATALOG if s["id"] not in [hook_stk["id"], outro_stk["id"]]]
                chosen = avail[len(placed_stickers) % len(avail)]
                placed_stickers.append({
                    "sticker": chosen,
                    "start_time": round(curr_t, 1),
                    "end_time": round(curr_t + 3.8, 1),
                    "position": "top-left" if len(placed_stickers) % 2 == 0 else "top-right"
                })
                curr_t += mid_interval

        return placed_stickers
